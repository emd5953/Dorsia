function evlog(string $s): void { file_put_contents('/app/storage/logs/eval.log', $s."\n", FILE_APPEND); }

function evdir(): string { $d = sys_get_temp_dir().'/evdone-'.bin2hex(random_bytes(6)); @mkdir($d); return $d; }

/** monitor closure: returns [checks, violations] */
function evmonitor(string $dir, int $n, int $slotId): Closure
{
    return function () use ($dir, $n, $slotId): array {
        $checks = 0; $viol = 0; $deadline = microtime(true) + 25;
        do {
            $r = DB::selectOne("select s.held_count, s.booked_count, s.capacity, (select count(*) from holds h where h.slot_id=s.id and h.status='active') as act from slots s where s.id=?", [$slotId]);
            $checks++;
            if ($r->held_count + $r->booked_count > $r->capacity || $r->held_count != $r->act || $r->held_count < 0) { $viol++; }
            $done = count(glob("$dir/d-*") ?: []);
        } while ($done < $n && microtime(true) < $deadline);
        return ['checks' => $checks, 'viol' => $viol];
    };
}

function evhold(int $memberId, int $slotId, ?string $key = null, ?callable $in = null): string
{
    try {
        app(HoldSlot::class)->handle(new HoldSlotInput($memberId, $slotId, 2, $key ?? (string) Str::uuid()), $in);
        return 'ok';
    } catch (Throwable $e) {
        if (! $e instanceof App\Shared\Exceptions\DomainError) { evlog('ERR '.get_class($e).': '.substr($e->getMessage(),0,300).' @ '.$e->getFile().':'.$e->getLine()); }
        return class_basename($e);
    }
}

function evrace(string $dir, int $slotId, array $closures, string $tag): array
{
    $n = count($closures);
    $wrapped = [];
    foreach ($closures as $i => $c) {
        $wrapped[] = function () use ($c, $dir, $i) { try { return $c(); } finally { touch("$dir/d-$i"); } };
    }
    $wrapped[] = evmonitor($dir, $n, $slotId);
    $res = Concurrency::run($wrapped);
    $mon = array_pop($res);
    $t = array_count_values(array_map(fn ($x) => is_string($x) ? $x : json_encode($x), $res));
    ksort($t);
    $s = DB::table('slots')->find($slotId);
    $act = Hold::query()->where('slot_id', $slotId)->where('status', 'active')->count();
    $gate = app(SlotGate::class)->value($slotId);
    evlog(sprintf('%s: results=%s held=%d booked=%d cap=%d active=%d gate=%s monitor=%d checks/%d viol', $tag, json_encode($t), $s->held_count, $s->booked_count, $s->capacity, $act, var_export($gate, true), $mon['checks'], $mon['viol']));
    return ['t' => $t, 'slot' => $s, 'act' => $act, 'gate' => $gate, 'mon' => $mon];
}

function evmembers(int $n): array { return Member::factory()->count($n)->create()->pluck('id')->all(); }

function gen(int $slotId): int { return (int) Redis::connection()->command('get', ["slot_gate:{$slotId}:generation"]); }

    $cl = [];
    foreach ($holds as $i => $h) {
        $cl[] = fn () => $i % 2 ? (function () use ($h) { try { DB::transaction(fn () => app(ConvertHoldToBooking::class)->handle($h->id, $h->memberId)); return 'conv'; } catch (Throwable $e) { return class_basename($e); } })()

it('R4 redis flush during refresh', function () {
    $slot = Fixtures::slot(['capacity' => 10]);
    foreach (range(1, 10) as $_) { Fixtures::hold($slot); }
    DB::table('holds')->where('slot_id', $slot->id)->limit(5)->update(['expires_at' => now()->subSecond()]);
    $m = evmembers(50);
    $cl = array_map(fn ($id) => fn () => evhold($id, $slot->id), $m);
    $cl[] = function () { for ($i = 0; $i < 10; $i++) { usleep(5000); Redis::connection()->flushdb(); } return 'flusher'; };
    $r = evrace(evdir(), $slot->id, $cl, 'R4');
    $truth = 10 - $r['slot']->held_count - $r['slot']->booked_count;
    evlog("R4 truth=$truth pending=".var_export(Redis::connection()->command('get', ["slot_gate:{$slot->id}:pending"]), true));
    expect($r['mon']['viol'])->toBe(0)->and($r['slot']->held_count)->toBeLessThanOrEqual(10);
    // free tables must be bookable within a window
    $t0 = microtime(true); $ok = 0; $last = '';
    while ($truth > 0 && microtime(true) - $t0 < 12) { $last = evhold(Fixtures::member()->id, $slot->id); if ($last === 'ok') { $ok++; $truth--; } else { usleep(250000); } }
    evlog(sprintf('R4 post: booked %d more in %.2fs last=%s', $ok, microtime(true) - $t0, $last));
    expect($truth)->toBe(0);
});
